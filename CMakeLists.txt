cmake_minimum_required(VERSION 3.24)
project(StereoDepth VERSION 0.3.1 LANGUAGES CXX)
option(S3D_DISTRIBUTABLE "Require an explicit runtime location instead of the build checkout" OFF)
if(S3D_DISTRIBUTABLE)
  set(S3D_RUNTIME_FALLBACK "")
else()
  set(S3D_RUNTIME_FALLBACK "${PROJECT_SOURCE_DIR}")
endif()
option(S3D_BUILD_PREMIERE "Build Adobe Premiere effect (requires licensed SDKs)" OFF)
option(S3D_BUILD_CUDA "Build experimental CUDA kernels" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(stereodepth STATIC
  native/src/depth.cpp native/src/temporal.cpp native/src/reprojection.cpp
  native/src/reconstruction.cpp native/src/content_fill.cpp native/src/packing.cpp native/src/pipeline.cpp
  native/src/cache_reader.cpp native/src/pfm.cpp)
target_include_directories(stereodepth PUBLIC native/include)
set_target_properties(stereodepth PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(stereodepth PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(stereodepth PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(s3d_render native/tools/render.cpp)
target_link_libraries(s3d_render PRIVATE stereodepth)
include(CTest)
if(BUILD_TESTING)
  add_executable(s3d_tests native/tests/test_core.cpp)
  target_link_libraries(s3d_tests PRIVATE stereodepth)
  add_test(NAME core COMMAND s3d_tests)
  add_executable(s3d_buffer_pool_tests native/gpu/test_buffer_pool.cpp)
  add_test(NAME gpu_buffer_ownership COMMAND s3d_buffer_pool_tests)
endif()
if(S3D_BUILD_CUDA)
  if(APPLE)
    message(FATAL_ERROR "CUDA is unavailable on Apple silicon; build the CPU stereo core with S3D_BUILD_CUDA=OFF")
  endif()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "86;89;120" CACHE STRING "CUDA targets: RTX 30, 40 and 50 series")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit 12.8 REQUIRED)
  add_library(stereodepth_cuda STATIC native/gpu/cuda_backend.cu)
  target_include_directories(stereodepth_cuda PUBLIC native/include)
  target_link_libraries(stereodepth_cuda PUBLIC CUDA::cudart)
  set_target_properties(stereodepth_cuda PROPERTIES CUDA_STANDARD 17)
endif()
if(APPLE)
  enable_language(OBJCXX)
  find_package(CURL REQUIRED)
  configure_file(native/premiere/runtime_config.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/runtime_config.hpp" @ONLY)
  add_library(s3d_http STATIC native/premiere/http_client_curl.cpp)
  target_link_libraries(s3d_http PUBLIC CURL::libcurl)
  add_library(s3d_bridge STATIC native/premiere/platform_mac.mm native/premiere/depth_bridge.cpp)
  target_include_directories(s3d_bridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_options(s3d_bridge PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  set_target_properties(s3d_bridge PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED ON)
  target_link_libraries(s3d_bridge PUBLIC stereodepth s3d_http "-framework AppKit")
  if(BUILD_TESTING)
    add_executable(s3d_http_tests native/premiere/test_http.cpp)
    target_link_libraries(s3d_http_tests PRIVATE s3d_http)
  endif()
endif()
if(S3D_BUILD_PREMIERE)
  add_subdirectory(native/premiere)
endif()
